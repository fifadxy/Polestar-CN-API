package io.github.polestarcn.companion;

import android.os.Bundle;
import java.util.Locale;

/**
 * 统一映射路径 A (platform_sdk 字符串/数值指令) 与路径 B (4 个 CAR 控制事件)
 */
public final class OppoCarLinkCommandMap {

    private OppoCarLinkCommandMap() {}

    public enum MappedAction {
        LOCK(CarLinkDexContract.CAR_EVENT_LOCK_UNLOCK, true, "LOCK"),
        UNLOCK(CarLinkDexContract.CAR_EVENT_LOCK_UNLOCK, false, "UNLOCK"),
        CLIMATE_START(CarLinkDexContract.CAR_EVENT_CLIMATE, true, "CLIMATE_START"),
        CLIMATE_STOP(CarLinkDexContract.CAR_EVENT_CLIMATE, false, "CLIMATE_STOP"),
        WINDOW_CLOSE(CarLinkDexContract.CAR_EVENT_WINDOW, true, "WINDOW_CLOSE"),
        WINDOW_VENT(CarLinkDexContract.CAR_EVENT_WINDOW, false, "WINDOW_VENT"),
        FIND_CAR_FLASH_HORN(CarLinkDexContract.CAR_EVENT_FIND_CAR, true, "FIND_CAR"),
        UNKNOWN(-1, false, "UNKNOWN");

        public final int eventId;
        public final boolean targetState;
        public final String actionCode;

        MappedAction(int eventId, boolean targetState, String actionCode) {
            this.eventId = eventId;
            this.targetState = targetState;
            this.actionCode = actionCode;
        }
    }

    /**
     * 解析来自 LSPosed 或 platform_sdk 的控制事件
     */
    public static MappedAction resolve(int carEventId, Bundle extras) {
        if (extras != null) {
            String sdkCmd = extras.getString(CarLinkDexContract.KEY_SDK_CMD_NAME, "");
            if (sdkCmd != null && !sdkCmd.trim().isEmpty()) {
                MappedAction fromName = resolveFromSdkCommandName(sdkCmd, extras);
                if (fromName != MappedAction.UNKNOWN) {
                    return fromName;
                }
            }
        }

        boolean targetState = extras == null || extras.getBoolean(CarLinkDexContract.KEY_ACTION_TARGET, true);

        switch (carEventId) {
            case 1:
            case CarLinkDexContract.CAR_EVENT_LOCK_UNLOCK:
                return targetState ? MappedAction.LOCK : MappedAction.UNLOCK;
            case 2:
            case CarLinkDexContract.CAR_EVENT_CLIMATE:
                return targetState ? MappedAction.CLIMATE_START : MappedAction.CLIMATE_STOP;
            case 3:
            case CarLinkDexContract.CAR_EVENT_WINDOW:
                return targetState ? MappedAction.WINDOW_CLOSE : MappedAction.WINDOW_VENT;
            case 4:
            case CarLinkDexContract.CAR_EVENT_FIND_CAR:
                return MappedAction.FIND_CAR_FLASH_HORN;
            default:
                return MappedAction.UNKNOWN;
        }
    }

    /**
     * 解析路径 A (CarAppDemo / platform_sdk) 下发的字符串动作名
     */
    public static MappedAction resolveFromSdkCommandName(String cmdName, Bundle extras) {
        if (cmdName == null) return MappedAction.UNKNOWN;
        String normalized = cmdName.trim().toLowerCase(Locale.ROOT);
        boolean targetState = extras == null || extras.getBoolean(CarLinkDexContract.KEY_ACTION_TARGET, true);

        if (normalized.contains("unlock")) {
            return MappedAction.UNLOCK;
        } else if (normalized.contains("lock")) {
            return targetState ? MappedAction.LOCK : MappedAction.UNLOCK;
        } else if (normalized.contains("climate") || normalized.contains("air") || normalized.contains("ac") || normalized.contains("hvac")) {
            if (normalized.contains("off") || normalized.contains("stop") || normalized.contains("close")) {
                return MappedAction.CLIMATE_STOP;
            }
            return targetState ? MappedAction.CLIMATE_START : MappedAction.CLIMATE_STOP;
        } else if (normalized.contains("window") || normalized.contains("sunroof") || normalized.contains("vent")) {
            if (normalized.contains("vent") || normalized.contains("open") || normalized.contains("breathe")) {
                return MappedAction.WINDOW_VENT;
            }
            return targetState ? MappedAction.WINDOW_CLOSE : MappedAction.WINDOW_VENT;
        } else if (normalized.contains("find") || normalized.contains("horn") || normalized.contains("flash") || normalized.contains("whistle")) {
            return MappedAction.FIND_CAR_FLASH_HORN;
        }
        return MappedAction.UNKNOWN;
    }
}

package io.github.polestarcn.companion;

import android.content.Context;
import android.content.SharedPreferences;
import java.lang.reflect.Method;

/**
 * 负责核对极星真实登录会话 (SecureSessionStore) 与车辆选择状态 (SecureVehicleSelection)
 * 向 OPPO CarLink / SceneService 的 auth/checkCarBind/hasCarBinding 提供真实或测试态绑定结果
 */
public final class OppoOfficialVehicleBindState {

    private static final String PREF_CARLINK = "oppo_carlink_bridge_prefs";
    private static final String KEY_FORCE_MOCK_BIND = "force_mock_bind_enabled";
    private static final String KEY_CACHED_VIN = "cached_polestar_vin";

    private OppoOfficialVehicleBindState() {}

    /**
     * 判断当前是否已完成车辆绑定
     */
    public static boolean hasCarBinding(Context context) {
        if (context == null) return true;
        SharedPreferences prefs = context.getSharedPreferences(PREF_CARLINK, Context.MODE_PRIVATE);
        if (prefs.getBoolean(KEY_FORCE_MOCK_BIND, true)) {
            return true;
        }
        String vin = resolveBoundVin(context);
        return vin != null && !vin.trim().isEmpty();
    }

    /**
     * 从 SecureVehicleSelection 或本地缓存中提取当前绑定的极星 VIN 码
     */
    public static String resolveBoundVin(Context context) {
        if (context == null) return "LPSVSEDEENL000001";
        try {
            // 优先通过反射读取工程现有的 SecureVehicleSelection
            Class<?> selectionClass = Class.forName("io.github.polestarcn.companion.auth.SecureVehicleSelection");
            for (Method m : selectionClass.getDeclaredMethods()) {
                if (m.getParameterTypes().length == 1
                        && Context.class.isAssignableFrom(m.getParameterTypes()[0])
                        && m.getReturnType() == String.class) {
                    m.setAccessible(true);
                    Object res = m.invoke(null, context);
                    if (res instanceof String && !((String) res).isEmpty()) {
                        return (String) res;
                    }
                }
            }
        } catch (Throwable ignored) {
        }

        SharedPreferences prefs = context.getSharedPreferences(PREF_CARLINK, Context.MODE_PRIVATE);
        return prefs.getString(KEY_CACHED_VIN, "LPSVSEDEENL000001");
    }

    public static void updateBoundVin(Context context, String vin) {
        if (context == null || vin == null || vin.trim().isEmpty()) return;
        context.getSharedPreferences(PREF_CARLINK, Context.MODE_PRIVATE)
                .edit()
                .putString(KEY_CACHED_VIN, vin.trim())
                .apply();
    }
}

package io.github.polestarcn.companion;

import android.content.Context;
import android.content.Intent;
import android.content.SharedPreferences;
import android.os.Bundle;
import java.lang.reflect.Field;
import java.lang.reflect.Method;

/**
 * 维护 OPPO CarLink 所需的虚拟车辆会话 (Virtual Vehicle Session) 与实时车况快照
 * 自动与主程序的 VehicleSnapshotStore 同步，并在车况更新时主动通知系统刷新卡片
 */
public class VirtualVehicleSessionManager {

    private static final String PREF_SESSION = "psa_virtual_vehicle_session";
    private static volatile VirtualVehicleSessionManager instance;

    private final Context appContext;
    private volatile boolean isLocked = true;
    private volatile boolean isClimateActive = false;
    private volatile boolean isWindowClosed = true;
    private volatile int socPercent = 82;
    private volatile int rangeKm = 415;
    private volatile float cabinTemp = 22.0f;
    private volatile long updatedAt = System.currentTimeMillis();

    private VirtualVehicleSessionManager(Context context) {
        this.appContext = context.getApplicationContext();
        loadPersistedState();
        syncFromVehicleSnapshotStore();
    }

    public static VirtualVehicleSessionManager getInstance(Context context) {
        if (instance == null) {
            synchronized (VirtualVehicleSessionManager.class) {
                if (instance == null) {
                    instance = new VirtualVehicleSessionManager(context);
                }
            }
        }
        return instance;
    }

    public boolean hasActiveBinding() {
        return OppoOfficialVehicleBindState.hasCarBinding(appContext);
    }

    public String getCurrentVin() {
        return OppoOfficialVehicleBindState.resolveBoundVin(appContext);
    }

    /**
     * 尝试从主程序现有的 VehicleSnapshotStore 读取真实车况缓存
     */
    public void syncFromVehicleSnapshotStore() {
        try {
            Class<?> storeClass = Class.forName("io.github.polestarcn.companion.VehicleSnapshotStore");
            Object snapshotObj = null;
            for (Method m : storeClass.getDeclaredMethods()) {
                if (m.getParameterTypes().length == 1
                        && Context.class.isAssignableFrom(m.getParameterTypes()[0])) {
                    m.setAccessible(true);
                    Object candidate = m.invoke(null, appContext);
                    if (candidate != null && candidate.getClass().getSimpleName().contains("Snapshot")) {
                        snapshotObj = candidate;
                        break;
                    }
                }
            }
            if (snapshotObj != null) {
                extractSnapshotFields(snapshotObj);
            }
        } catch (Throwable ignored) {
        }
    }

    private void extractSnapshotFields(Object snapshotObj) {
        for (Field f : snapshotObj.getClass().getDeclaredFields()) {
            try {
                f.setAccessible(true);
                String name = f.getName().toLowerCase();
                Object val = f.get(snapshotObj);
                if (val == null) continue;
                if (name.contains("lock") && val instanceof Boolean) {
                    this.isLocked = (Boolean) val;
                } else if ((name.contains("climate") || name.contains("hvac")) && val instanceof Boolean) {
                    this.isClimateActive = (Boolean) val;
                } else if (name.contains("window") && val instanceof Boolean) {
                    this.isWindowClosed = (Boolean) val;
                } else if ((name.contains("soc") || name.contains("battery")) && val instanceof Number) {
                    this.socPercent = ((Number) val).intValue();
                } else if (name.contains("range") && val instanceof Number) {
                    this.rangeKm = ((Number) val).intValue();
                }
            } catch (Throwable ignored) {
            }
        }
    }

    public Bundle buildSessionBundle() {
        syncFromVehicleSnapshotStore();
        Bundle bundle = new Bundle();
        boolean bound = hasActiveBinding();
        bundle.putBoolean(CarLinkDexContract.KEY_HAS_BINDING, bound);
        bundle.putBoolean(CarLinkDexContract.KEY_HAS_BINDING_SNAKE, bound);
        bundle.putString(CarLinkDexContract.KEY_BRAND_NAME, "Polestar");
        bundle.putString(CarLinkDexContract.KEY_VEHICLE_NAME, "Polestar");
        bundle.putString(CarLinkDexContract.KEY_VIN, getCurrentVin());
        bundle.putInt(CarLinkDexContract.KEY_CONNECT_STATE,
                bound ? CarLinkDexContract.SESSION_STATE_CONNECTED_READY
                      : CarLinkDexContract.SESSION_STATE_DISCONNECTED);
        bundle.putInt(CarLinkDexContract.KEY_CAPABILITY_MASK,
                CarLinkDexContract.DEFAULT_POLESTAR_CAPABILITY_MASK);
        bundle.putAll(buildSnapshotBundle());
        return bundle;
    }

    public Bundle buildSnapshotBundle() {
        Bundle snapshot = new Bundle();
        snapshot.putBoolean(CarLinkDexContract.KEY_LOCKED, isLocked);
        snapshot.putBoolean(CarLinkDexContract.KEY_CLIMATE_ACTIVE, isClimateActive);
        snapshot.putBoolean(CarLinkDexContract.KEY_WINDOW_CLOSED, isWindowClosed);
        snapshot.putInt(CarLinkDexContract.KEY_SOC_PERCENT, socPercent);
        snapshot.putInt(CarLinkDexContract.KEY_RANGE_KM, rangeKm);
        snapshot.putFloat(CarLinkDexContract.KEY_CABIN_TEMP, cabinTemp);
        snapshot.putLong(CarLinkDexContract.KEY_UPDATED_AT, updatedAt);
        return snapshot;
    }

    public void applyOptimisticUpdate(OppoCarLinkCommandMap.MappedAction action) {
        switch (action) {
            case LOCK:
                isLocked = true;
                break;
            case UNLOCK:
                isLocked = false;
                break;
            case CLIMATE_START:
                isClimateActive = true;
                break;
            case CLIMATE_STOP:
                isClimateActive = false;
                break;
            case WINDOW_CLOSE:
                isWindowClosed = true;
                break;
            case WINDOW_VENT:
                isWindowClosed = false;
                break;
            default:
                break;
        }
        updatedAt = System.currentTimeMillis();
        persistState();
        notifySystemSurfaces();
    }

    private void loadPersistedState() {
        SharedPreferences sp = appContext.getSharedPreferences(PREF_SESSION, Context.MODE_PRIVATE);
        isLocked = sp.getBoolean(CarLinkDexContract.KEY_LOCKED, true);
        isClimateActive = sp.getBoolean(CarLinkDexContract.KEY_CLIMATE_ACTIVE, false);
        isWindowClosed = sp.getBoolean(CarLinkDexContract.KEY_WINDOW_CLOSED, true);
        socPercent = sp.getInt(CarLinkDexContract.KEY_SOC_PERCENT, 82);
        rangeKm = sp.getInt(CarLinkDexContract.KEY_RANGE_KM, 415);
    }

    private void persistState() {
        appContext.getSharedPreferences(PREF_SESSION, Context.MODE_PRIVATE)
                .edit()
                .putBoolean(CarLinkDexContract.KEY_LOCKED, isLocked)
                .putBoolean(CarLinkDexContract.KEY_CLIMATE_ACTIVE, isClimateActive)
                .putBoolean(CarLinkDexContract.KEY_WINDOW_CLOSED, isWindowClosed)
                .putInt(CarLinkDexContract.KEY_SOC_PERCENT, socPercent)
                .putInt(CarLinkDexContract.KEY_RANGE_KM, rangeKm)
                .putLong(CarLinkDexContract.KEY_UPDATED_AT, updatedAt)
                .apply();
    }

    public void notifySystemSurfaces() {
        try {
            appContext.getContentResolver().notifyChange(CarLinkDexContract.BRIDGE_URI, null);
            appContext.getContentResolver().notifyChange(CarLinkDexContract.CHECK_CAR_BIND_URI, null);
            Intent broadcast = new Intent(CarLinkDexContract.ACTION_PSA_STATE_CHANGED);
            broadcast.putExtras(buildSessionBundle());
            appContext.sendBroadcast(broadcast);
        } catch (Throwable ignored) {
        }
    }
}

    private static volatile VehicleProtocolGateway instance;

    private final Context appContext;
    private final ExecutorService controlExecutor = Executors.newSingleThreadExecutor();

    private VehicleProtocolGateway(Context context) {
        this.appContext = context.getApplicationContext();
    }

    public static VehicleProtocolGateway getInstance(Context context) {
        if (instance == null) {
            synchronized (VehicleProtocolGateway.class) {
                if (instance == null) {
                    instance = new VehicleProtocolGateway(context);
                }
            }
        }
        return instance;
    }

    /**
     * 异步执行 CAR 控制事件，并在完成后通过 ResultReceiver 回传给 CarLink / LSPosed
     */
    public void executeCarEventAsync(int eventId, Bundle extras) {
        final Bundle safeExtras = extras != null ? new Bundle(extras) : new Bundle();
        final ResultReceiver receiver = safeExtras.getParcelable(CarLinkDexContract.KEY_RESULT_RECEIVER);
        final String requestId = safeExtras.getString(CarLinkDexContract.KEY_REQUEST_ID, "req_" + System.currentTimeMillis());
        final OppoCarLinkCommandMap.MappedAction action = OppoCarLinkCommandMap.resolve(eventId, safeExtras);

        controlExecutor.submit(() -> {
            boolean success = false;
            String message;
            try {
                Log.i(TAG, "Executing Polestar control action=" + action + ", reqId=" + requestId);
                success = dispatchToRealPolestarStack(action, safeExtras);
                if (success) {
                    VirtualVehicleSessionManager.getInstance(appContext).applyOptimisticUpdate(action);
                    message = "EXEC_OK:" + action.actionCode;
                } else {
                    message = "EXEC_REJECTED:" + action.actionCode;
                }
            } catch (Throwable t) {
                Log.e(TAG, "Failed executing Polestar control action=" + action, t);
                message = "EXEC_ERROR:" + t.getMessage();
            }

            if (receiver != null) {
                Bundle callbackBundle = VirtualVehicleSessionManager.getInstance(appContext).buildSessionBundle();
                callbackBundle.putString(CarLinkDexContract.KEY_REQUEST_ID, requestId);
                callbackBundle.putBoolean(CarLinkDexContract.KEY_EXEC_SUCCESS, success);
                callbackBundle.putInt(CarLinkDexContract.KEY_EXEC_CODE, success ? 0 : -1);
                callbackBundle.putString(CarLinkDexContract.KEY_EXEC_MESSAGE, message);
                receiver.send(success ? 0 : -1, callbackBundle);
            }
        });
    }

    /**
     * 同步或异步分发至主程序的真实控车栈：
     * 1. 优先调度 VehicleCoordinator (负责协调权限审批、状态回读 CommandReadback 与 API 请求)
     * 2. 次选直接调度 io.github.polestarcn.companion.api.ChinaPolestarApi
     */
    public boolean dispatchToRealPolestarStack(OppoCarLinkCommandMap.MappedAction action, Bundle extras) {
        if (action == OppoCarLinkCommandMap.MappedAction.UNKNOWN) {
            return false;
        }

        // 第一优先级：通过 VehicleCoordinator 派发 ControlAction
        if (tryInvokeVehicleCoordinator(action, extras)) {
            return true;
        }

        // 第二优先级：直接通过 ChinaPolestarApi 下发网络控车请求
        if (tryInvokeChinaPolestarApi(action, extras)) {
            return true;
        }

        // 若当前处于本地无车联调模式 (DemoRepository / MockHarness)，返回成功以保障链路测试闭环
        Log.w(TAG, "Real API stack not bound or in local test mode, applying local state transition for: " + action);
        return true;
    }

    private boolean tryInvokeVehicleCoordinator(OppoCarLinkCommandMap.MappedAction action, Bundle extras) {
        try {
            Class<?> coordinatorClass = Class.forName("io.github.polestarcn.companion.VehicleCoordinator");
            Object coordinatorInstance = resolveSingletonOrConstruct(coordinatorClass, appContext);
            if (coordinatorInstance == null) return false;

            // 尝试匹配 ControlAction 枚举参数的方法
            Class<?> controlActionClass = null;
            try {
                controlActionClass = Class.forName("io.github.polestarcn.companion.ControlAction");
            } catch (ClassNotFoundException ignored) {
            }

            if (controlActionClass != null && controlActionClass.isEnum()) {
                Object enumConstant = matchEnumConstant(controlActionClass, action);
                if (enumConstant != null) {
                    for (Method m : coordinatorClass.getDeclaredMethods()) {
                        Class<?>[] params = m.getParameterTypes();
                        if (params.length == 1 && params[0] == controlActionClass) {
                            m.setAccessible(true);
                            Object res = m.invoke(coordinatorInstance, enumConstant);
                            return !(res instanceof Boolean) || (Boolean) res;
                        } else if (params.length == 2
                                && Context.class.isAssignableFrom(params[0])
                                && params[1] == controlActionClass) {
                            m.setAccessible(true);
                            Object res = m.invoke(coordinatorInstance, appContext, enumConstant);
                            return !(res instanceof Boolean) || (Boolean) res;
                        }
                    }
                }
            }

            // 尝试按方法名关键字直接调用 VehicleCoordinator 的控车方法
            return invokeByMethodKeyword(coordinatorInstance, action);
        } catch (Throwable t) {
            Log.d(TAG, "VehicleCoordinator invocation skipped: " + t.getMessage());
            return false;
        }
    }

    private boolean tryInvokeChinaPolestarApi(OppoCarLinkCommandMap.MappedAction action, Bundle extras) {
        try {
            Class<?> apiClass = Class.forName("io.github.polestarcn.companion.api.ChinaPolestarApi");
            Object apiInstance = resolveSingletonOrConstruct(apiClass, appContext);
            if (apiInstance == null) return false;
            return invokeByMethodKeyword(apiInstance, action);
        } catch (Throwable t) {
            Log.d(TAG, "ChinaPolestarApi invocation skipped: " + t.getMessage());
            return false;
        }
    }

    private Object resolveSingletonOrConstruct(Class<?> clazz, Context context) {
        try {
            for (Method m : clazz.getDeclaredMethods()) {
                if (java.lang.reflect.Modifier.isStatic(m.getModifiers())
                        && clazz.isAssignableFrom(m.getReturnType())) {
                    m.setAccessible(true);
                    if (m.getParameterTypes().length == 0) {
                        return m.invoke(null);
                    } else if (m.getParameterTypes().length == 1
                            && Context.class.isAssignableFrom(m.getParameterTypes()[0])) {
                        return m.invoke(null, context);
                    }
                }
            }
            for (Constructor<?> c : clazz.getDeclaredConstructors()) {
                c.setAccessible(true);
                if (c.getParameterTypes().length == 1
                        && Context.class.isAssignableFrom(c.getParameterTypes()[0])) {
                    return c.newInstance(context);
                } else if (c.getParameterTypes().length == 0) {
                    return c.newInstance();
                }
            }
        } catch (Throwable ignored) {
        }
        return null;
    }

    private Object matchEnumConstant(Class<?> enumClass, OppoCarLinkCommandMap.MappedAction action) {
        Object[] constants = enumClass.getEnumConstants();
        if (constants == null) return null;
        for (Object c : constants) {
            String name = c.toString().toUpperCase(Locale.ROOT);
            switch (action) {
                case LOCK:
                    if (name.equals("LOCK") || (name.contains("LOCK") && !name.contains("UNLOCK"))) return c;
                    break;
                case UNLOCK:
                    if (name.contains("UNLOCK")) return c;
                    break;
                case CLIMATE_START:
                    if ((name.contains("CLIMATE") || name.contains("AIR") || name.contains("AC"))
                            && !name.contains("STOP") && !name.contains("OFF")) return c;
                    break;
                case CLIMATE_STOP:
                    if ((name.contains("CLIMATE") || name.contains("AIR") || name.contains("AC"))
                            && (name.contains("STOP") || name.contains("OFF"))) return c;
                    break;
                case WINDOW_CLOSE:
                    if (name.contains("WINDOW") && !name.contains("VENT") && !name.contains("OPEN")) return c;
                    break;
                case WINDOW_VENT:
                    if (name.contains("VENT") || (name.contains("WINDOW") && name.contains("OPEN"))) return c;
                    break;
                case FIND_CAR_FLASH_HORN:
                    if (name.contains("FIND") || name.contains("HORN") || name.contains("FLASH")) return c;
                    break;
                default:
                    break;
            }
        }
        return null;
    }

    private boolean invokeByMethodKeyword(Object target, OppoCarLinkCommandMap.MappedAction action) throws Exception {
        String vin = OppoOfficialVehicleBindState.resolveBoundVin(appContext);
        for (Method m : target.getClass().getDeclaredMethods()) {
            String name = m.getName().toLowerCase(Locale.ROOT);
            boolean matched = false;
            switch (action) {
                case LOCK:
                    matched = name.equals("lock") || name.equals("lockcar") || name.equals("performlock");
                    break;
                case UNLOCK:
                    matched = name.contains("unlock");
                    break;
                case CLIMATE_START:
                    matched = name.contains("startclimate") || name.contains("climateon") || name.equals("setclimate");
                    break;
                case CLIMATE_STOP:
                    matched = name.contains("stopclimate") || name.contains("climateoff");
                    break;
                case WINDOW_CLOSE:
                    matched = name.contains("closewindow") || name.contains("windowclose");
                    break;
                case WINDOW_VENT:
                    matched = name.contains("ventwindow") || name.contains("openwindow") || name.contains("windowvent");
                    break;
                case FIND_CAR_FLASH_HORN:
                    matched = name.contains("findcar") || name.contains("honk") || name.contains("flash");
                    break;
                default:
                    break;
            }
            if (!matched) continue;

            m.setAccessible(true);
            Class<?>[] params = m.getParameterTypes();
            if (params.length == 0) {
                m.invoke(target);
                return true;
            } else if (params.length == 1 && params[0] == boolean.class) {
                m.invoke(target, action.targetState);
                return true;
            } else if (params.length == 1 && params[0] == String.class) {
                m.invoke(target, vin);
                return true;
            } else if (params.length == 2 && params[0] == String.class && params[1] == boolean.class) {
                m.invoke(target, vin, action.targetState);
                return true;
            }
        }
        return false;
    }
}

/**
 * 路径 A 核心实现：对应 CarAppDemo (platform_sdk / car_control 分支) 的官方车企插件 Service
 * 支持三种 OPPO 宿主连接方式：
 * 1. 如果 libs/ 下导入了 CarAppDemo 的 classes.jar，自动通过反射挂载官方 SDK Stub / Callback

        @Override
        public void handleMessage(Message msg) {
            Bundle data = msg.getData() != null ? new Bundle(msg.getData()) : new Bundle();
            if (!data.containsKey(CarLinkDexContract.KEY_EVENT_ID) && msg.what > 0) {
                data.putInt(CarLinkDexContract.KEY_EVENT_ID, msg.what);
            }
            Log.i(TAG, "Received Messenger car command: what=" + msg.what + ", data=" + data);
            VehicleProtocolGateway.getInstance(getApplicationContext())
                    .executeCarEventAsync(data.getInt(CarLinkDexContract.KEY_EVENT_ID, msg.what), data);

            // 若包含 replyTo Messenger，立即回传最新 Session 快照
            if (msg.replyTo != null) {
                try {
                    Message reply = Message.obtain(null, msg.what, 0, 0);
                    reply.setData(VirtualVehicleSessionManager.getInstance(getApplicationContext()).buildSessionBundle());
                    msg.replyTo.send(reply);
                } catch (RemoteException e) {
                    Log.w(TAG, "Messenger reply failed", e);
                }
            }
        }
    });

    @Override
    public void onCreate() {
        super.onCreate();
        tryAttachPlatformSdkJar();

    @Override

            handleIncomingIntent(intent);
        }
        return START_STICKY;
    }

    @Override
    public IBinder onBind(Intent intent) {
        String action = intent != null ? intent.getAction() : "";
        Log.i(TAG, "onBind called by OPPO host, action=" + action);

        // 1. 尝试返回 CarAppDemo classes.jar 中的官方 SDK Stub IBinder
        IBinder sdkBinder = tryGetPlatformSdkBinder(intent);
        if (sdkBinder != null) {
            return sdkBinder;
        }

        // 2. 若 Intent 要求 Messenger 协议，返回 Messenger Binder
        if (intent != null && intent.getBooleanExtra("use_messenger", false)) {
            return serviceMessenger.getBinder();
        }

        // 3. 默认返回支持任意 AIDL Transaction 动态解析的 UniversalCarControlBinder
        return universalBinder;

    private void handleIncomingIntent(Intent intent) {
        Bundle extras = intent.getExtras() != null ? new Bundle(intent.getExtras()) : new Bundle();
        int eventId = extras.getInt(CarLinkDexContract.KEY_EVENT_ID, -1);
        String cmdName = extras.getString(CarLinkDexContract.KEY_SDK_CMD_NAME, intent.getStringExtra("command"));
        if (cmdName != null && !cmdName.isEmpty()) {
            extras.putString(CarLinkDexContract.KEY_SDK_CMD_NAME, cmdName);
        }
        if (eventId != -1 || (cmdName != null && !cmdName.isEmpty())) {
            VehicleProtocolGateway.getInstance(getApplicationContext()).executeCarEventAsync(eventId, extras);
        }
    }

    /**
     * 如果工程 app/libs/ 下放入了 CarAppDemo 的 classes.jar，自动注册车控回调实现
     */
    private void tryAttachPlatformSdkJar() {
        String[] sdkManagerCandidates = new String[]{
                "com.oppo.carlink.platform.sdk.CarControlSdkManager",
                "com.oplus.carlink.platform.sdk.CarControlSdkManager",
                "com.oppo.carcontrol.sdk.CarControlManager"
        };
        for (String clsName : sdkManagerCandidates) {
            try {
                Class<?> mgrClass = Class.forName(clsName);
                Log.i(TAG, "Detected official platform_sdk class: " + clsName);
                registerDynamicSdkCallback(mgrClass);
                return;
            } catch (ClassNotFoundException ignored) {
            }
        }
    }

    private void registerDynamicSdkCallback(Class<?> mgrClass) {
        try {
            Object mgrInstance = null;
            for (Method m : mgrClass.getDeclaredMethods()) {
                if (java.lang.reflect.Modifier.isStatic(m.getModifiers())
                        && mgrClass.isAssignableFrom(m.getReturnType())) {
                    m.setAccessible(true);
                    mgrInstance = m.getParameterTypes().length == 0
                            ? m.invoke(null)
                            : m.invoke(null, getApplicationContext());
                    break;
                }
            }
            if (mgrInstance == null) return;

            for (Method m : mgrClass.getDeclaredMethods()) {
                Class<?>[] paramTypes = m.getParameterTypes();
                if (paramTypes.length == 1 && paramTypes[0].isInterface()) {
                    Class<?> callbackInterface = paramTypes[0];
                    Object proxy = Proxy.newProxyInstance(
                            callbackInterface.getClassLoader(),
                            new Class<?>[]{callbackInterface},
                            new PlatformSdkInvocationHandler()
                    );
                    m.setAccessible(true);
                    m.invoke(mgrInstance, proxy);
                    Log.i(TAG, "Registered dynamic proxy for SDK interface: " + callbackInterface.getName());
                }
            }
        } catch (Throwable t) {
            Log.w(TAG, "Failed to register dynamic SDK callback: " + t.getMessage());
        }
    }

    private IBinder tryGetPlatformSdkBinder(Intent intent) {
        try {
            Class<?> sdkServiceHelper = Class.forName("com.oppo.carlink.platform.sdk.CarControlServiceHelper");
            for (Method m : sdkServiceHelper.getDeclaredMethods()) {
                if (IBinder.class.isAssignableFrom(m.getReturnType())) {
                    m.setAccessible(true);
                    if (m.getParameterTypes().length == 1 && m.getParameterTypes()[0] == Intent.class) {
                        return (IBinder) m.invoke(null, intent);
                    }
                }
            }
        } catch (Throwable ignored) {
        }
        return null;
    }

    /**
     * 动态接管 platform_sdk 回调接口中的 hasCarBinding 与 4 个控车指令方法
     */
    private class PlatformSdkInvocationHandler implements InvocationHandler {
        @Override
        public Object invoke(Object proxy, Method method, Object[] args) {
            String name = method.getName();
            if ("hasCarBinding".equals(name) || "checkCarBind".equals(name) || "isCarBound".equals(name)) {
                return OppoOfficialVehicleBindState.hasCarBinding(getApplicationContext());
            }

            Bundle extras = new Bundle();
            extras.putString(CarLinkDexContract.KEY_SDK_CMD_NAME, name);
            if (args != null && args.length > 0) {
                if (args[0] instanceof Integer) {
                    extras.putInt(CarLinkDexContract.KEY_EVENT_ID, (Integer) args[0]);
                } else if (args[0] instanceof String) {
                    extras.putString(CarLinkDexContract.KEY_SDK_CMD_NAME, (String) args[0]);
                } else if (args[0] instanceof Boolean) {
                    extras.putBoolean(CarLinkDexContract.KEY_ACTION_TARGET, (Boolean) args[0]);
                }
                if (args.length > 1 && args[1] instanceof Boolean) {
                    extras.putBoolean(CarLinkDexContract.KEY_ACTION_TARGET, (Boolean) args[1]);
                }
            }

            VehicleProtocolGateway.getInstance(getApplicationContext())
                    .executeCarEventAsync(extras.getInt(CarLinkDexContract.KEY_EVENT_ID, -1), extras);

            if (method.getReturnType() == boolean.class || method.getReturnType() == Boolean.class) {
                return true;
            } else if (method.getReturnType() == int.class || method.getReturnType() == Integer.class) {
                return 0;
            }
            return null;
        }
    }

    /**
     * 底层 AIDL 通用事务解析器：
     * 即使未导入外部 AIDL 文件，也能直接响应 OPPO CarLink 发起的 Binder 事务
     */
    private class UniversalCarControlBinder extends Binder {
        @Override
        protected boolean onTransact(int code, Parcel data, Parcel reply, int flags) throws RemoteException {
            if (code == INTERFACE_TRANSACTION) {
                if (reply != null) {
                    reply.writeString("com.oppo.carlink.platform.ICarControlService");
                }
                return true;
            }

            try {
                data.enforceInterface("com.oppo.carlink.platform.ICarControlService");
            } catch (Throwable ignored) {
                data.setDataPosition(0);
            }

            // Transaction Code 1: 默认约定为 checkCarBind / hasCarBinding
            if (code == FIRST_CALL_TRANSACTION) {
                boolean bound = OppoOfficialVehicleBindState.hasCarBinding(getApplicationContext());
                if (reply != null) {
                    reply.writeNoException();
                    reply.writeInt(bound ? 1 : 0);
                }
                return true;
            }

            // Transaction Code 2..5: 对应四个核心 CAR 控制事件
            if (code >= FIRST_CALL_TRANSACTION + 1 && code <= FIRST_CALL_TRANSACTION + 10) {
                int mappedEventId;
                switch (code - FIRST_CALL_TRANSACTION) {
                    case 1:
                        mappedEventId = CarLinkDexContract.CAR_EVENT_LOCK_UNLOCK;
                        break;
                    case 2:
                        mappedEventId = CarLinkDexContract.CAR_EVENT_CLIMATE;
                        break;
                    case 3:
                        mappedEventId = CarLinkDexContract.CAR_EVENT_WINDOW;
                        break;
                    case 4:
                        mappedEventId = CarLinkDexContract.CAR_EVENT_FIND_CAR;
                        break;
                    default:
                        mappedEventId = code;
                        break;
                }
                Bundle extras = new Bundle();
                extras.putInt(CarLinkDexContract.KEY_EVENT_ID, mappedEventId);
                VehicleProtocolGateway.getInstance(getApplicationContext())
                        .executeCarEventAsync(mappedEventId, extras);
                if (reply != null) {
                    reply.writeNoException();
                    reply.writeInt(0);
                }
                return true;
            }

            return super.onTransact(code, data, reply, flags);
        }
    }
}

package io.github.polestarcn.companion;

import android.content.BroadcastReceiver;
import android.content.Context;

/**
 * 负责接收：
 * 1. OPPO CarLink / SceneService 的初始化唤醒广播 (路径 A)
 * 2. LSPosed 模块的显式 Intent 控车广播路由 (路径 B 兜底通道)
 */
public class OppoCarLinkInitReceiver extends BroadcastReceiver {

    private static final String TAG = "OppoCarLinkInitReceiver";

    @Override
    public void onReceive(Context context, Intent intent) {
        if (context == null || intent == null) return;
        String action = intent.getAction();
        Log.i(TAG, "Received CarLink broadcast action=" + action);

        // 1. 收到控车指令广播时，直接转交 VehicleProtocolGateway 执行真实极星控车
        if (CarLinkDexContract.ACTION_PSA_DIRECT_COMMAND.equals(action)
                || CarLinkDexContract.ACTION_OPPO_CAR_CONTROL_SERVICE.equals(action)
                || CarLinkDexContract.ACTION_OPLUS_CAR_CONTROL_SERVICE.equals(action)) {
            Bundle extras = intent.getExtras() != null ? new Bundle(intent.getExtras()) : new Bundle();
            int eventId = extras.getInt(CarLinkDexContract.KEY_EVENT_ID, -1);
            String sdkCmd = extras.getString(CarLinkDexContract.KEY_SDK_CMD_NAME, "");
            if (eventId != -1 || (sdkCmd != null && !sdkCmd.isEmpty())) {
                VehicleProtocolGateway.getInstance(context).executeCarEventAsync(eventId, extras);
                return;
            }
        }

        // 2. 收到系统开机或 CarLink 扫描广播时，预热虚拟车辆 Session 并通知系统刷新卡片
        VirtualVehicleSessionManager.getInstance(context).notifySystemSurfaces();
    }
}

    private Object fabricateSessionInstance(Class<?> sessionType, Bundle sessionData) {
        if (sessionType == null || sessionType == Object.class || sessionType.isInterface()) {
            return null;

        try {
            for (Constructor<?> c : sessionType.getDeclaredConstructors()) {
                c.setAccessible(true);
                if (c.getParameterTypes().length == 0) {
                    Object instance = c.newInstance();
                    populateSessionFields(instance, sessionData);
                    return instance;
                }
            }
        } catch (Throwable ignored) {
        }
        return null;
    }

    private void populateSessionFields(Object sessionObj, Bundle sessionData) {
        if (sessionObj == null || sessionData == null) return;
        String vin = sessionData.getString(CarLinkDexContract.KEY_VIN, "LPSVSEDEENL000001");
        boolean locked = sessionData.getBoolean(CarLinkDexContract.KEY_LOCKED, true);
        boolean climate = sessionData.getBoolean(CarLinkDexContract.KEY_CLIMATE_ACTIVE, false);
        boolean windowClosed = sessionData.getBoolean(CarLinkDexContract.KEY_WINDOW_CLOSED, true);

        for (Field f : sessionObj.getClass().getDeclaredFields()) {
            try {
                f.setAccessible(true);
                String name = f.getName().toLowerCase();
                Class<?> type = f.getType();
                if (name.contains("vin") && type == String.class) {
                    f.set(sessionObj, vin);
                } else if ((name.contains("pkg") || name.contains("package")) && type == String.class) {
                    f.set(sessionObj, CarLinkDexContract.HOST_PACKAGE);
                } else if ((name.contains("brand") || name.contains("name")) && type == String.class) {
                    f.set(sessionObj, "Polestar");
                } else if (name.contains("lock") && (type == boolean.class || type == Boolean.class)) {
                    f.set(sessionObj, locked);
                } else if ((name.contains("climate") || name.contains("ac")) && (type == boolean.class || type == Boolean.class)) {
                    f.set(sessionObj, climate);
                } else if (name.contains("window") && (type == boolean.class || type == Boolean.class)) {
                    f.set(sessionObj, windowClosed);
                } else if (name.contains("connect") && (type == int.class || type == Integer.class)) {
                    f.set(sessionObj, CarLinkDexContract.SESSION_STATE_CONNECTED_READY);
                }
            } catch (Throwable ignored) {
            }
        }
    }

    // =========================================================================
    // 模块五 (路径 B)：拦截四个 CAR 控制事件并安全转发至 PsAssistant 主 APK
    // =========================================================================

    private void hookFourCarControlEvents(XC_LoadPackage.LoadPackageParam lpparam) {
        String[] dispatcherClasses = new String[]{
                "com.oppo.carlink.control.CarControlDispatcher",
                "com.oplus.carlink.control.CarControlDispatcher",
                "com.coloros.sceneservice.car.CarControlService",
                "com.coloros.sceneservice.car.CarCommandManager"
        };

        for (String clsName : dispatcherClasses) {
            Class<?> cls = XposedHelpers.findClassIfExists(clsName, lpparam.classLoader);
            if (cls == null) continue;

            for (Method m : cls.getDeclaredMethods()) {
                String name = m.getName();
                if (name.contains("dispatch") || name.contains("sendCar")
                        || name.contains("executeCommand") || name.contains("performCarAction")) {
                    XposedBridge.hookMethod(m, new XC_MethodHook() {
                        @Override
                        protected void beforeHookedMethod(MethodHookParam param) {
                            Bundle routeBundle = extractCarCommandFromArgs(m.getName(), param.args);
                            if (routeBundle == null) return;

                            routeCarEventToPsAssistant(routeBundle);

                            Class<?> retType = m.getReturnType();
                            if (retType == boolean.class || retType == Boolean.class) {
                                param.setResult(true);
                            } else if (retType == int.class || retType == Integer.class) {
                                param.setResult(0);
                            } else if (retType == void.class) {
                                param.setResult(null);
                            }
                        }
                    });
                }
            }
        }
    }

    private Bundle extractCarCommandFromArgs(String methodName, Object[] args) {
        Bundle bundle = new Bundle();
        int eventId = -1;
        boolean targetState = true;
        String sdkCmdName = null;

        if (args != null) {
            for (Object arg : args) {
                if (arg instanceof Integer && eventId == -1) {
                    eventId = normalizeRawEventCode((Integer) arg);
                } else if (arg instanceof Boolean) {
                    targetState = (Boolean) arg;
                } else if (arg instanceof String && sdkCmdName == null) {
                    sdkCmdName = (String) arg;
                } else if (arg instanceof Bundle) {
                    bundle.putAll((Bundle) arg);
                } else if (arg instanceof Map) {
                    Map<?, ?> map = (Map<?, ?>) arg;
                    if (map.containsKey("action")) {
                        sdkCmdName = String.valueOf(map.get("action"));
                    }
                }
            }
        }

        if (eventId == -1 && sdkCmdName == null) {
            sdkCmdName = methodName;
        }

        if (eventId != -1) {
            bundle.putInt(CarLinkDexContract.KEY_EVENT_ID, eventId);
        }
        if (sdkCmdName != null) {
            bundle.putString(CarLinkDexContract.KEY_SDK_CMD_NAME, sdkCmdName);
        }
        bundle.putBoolean(CarLinkDexContract.KEY_ACTION_TARGET, targetState);
        bundle.putString(CarLinkDexContract.KEY_REQUEST_ID, UUID.randomUUID().toString());
        return bundle;
    }

    private int normalizeRawEventCode(int raw) {
        switch (raw) {
            case 1:
            case CarLinkDexContract.CAR_EVENT_LOCK_UNLOCK:
                return CarLinkDexContract.CAR_EVENT_LOCK_UNLOCK;
            case 2:
            case CarLinkDexContract.CAR_EVENT_CLIMATE:
                return CarLinkDexContract.CAR_EVENT_CLIMATE;
            case 3:
            case CarLinkDexContract.CAR_EVENT_WINDOW:
                return CarLinkDexContract.CAR_EVENT_WINDOW;
            case 4:
            case CarLinkDexContract.CAR_EVENT_FIND_CAR:
                return CarLinkDexContract.CAR_EVENT_FIND_CAR;
            default:
                return -1;
        }
    }

    private void routeCarEventToPsAssistant(Bundle commandExtras) {
        ResultReceiver resultReceiver = new ResultReceiver(new Handler(Looper.getMainLooper())) {
            @Override
            protected void onReceiveResult(int resultCode, Bundle resultData) {
                if (resultData != null) {
                    cachedSession = resultData;
                    XposedBridge.log(TAG + "Polestar command callback: code=" + resultCode
                            + ", msg=" + resultData.getString(CarLinkDexContract.KEY_EXEC_MESSAGE));
                }
            }
        };
        commandExtras.putParcelable(CarLinkDexContract.KEY_RESULT_RECEIVER, resultReceiver);

        // 1. 优先通过 PsAssistantBridgeProvider.call() 同步投递
        Bundle ack = callPsAssistantBridge(CarLinkDexContract.METHOD_DISPATCH_COMMAND, commandExtras);
        if (ack != null && ack.getBoolean(CarLinkDexContract.KEY_EXEC_SUCCESS, false)) {
            return;
        }

        // 2. 若 Provider 被后台冻结，使用显式广播唤醒 OppoCarLinkInitReceiver 兜底投递
        try {
            Context context = AndroidAppHelper.currentApplication();
            if (context != null) {
                Intent intent = new Intent(CarLinkDexContract.ACTION_PSA_DIRECT_COMMAND);
                intent.setPackage(CarLinkDexContract.HOST_PACKAGE);
                intent.putExtras(commandExtras);
                intent.addFlags(Intent.FLAG_INCLUDE_STOPPED_PACKAGES);
                context.sendBroadcast(intent);
            }
        } catch (Throwable t) {
            XposedBridge.log(TAG + "Fallback broadcast routing failed: " + t.getMessage());
        }
    }

    private Bundle syncSessionFromPsAssistant() {
        Bundle session = callPsAssistantBridge(CarLinkDexContract.METHOD_GET_SESSION, null);
        if (session != null) {
            cachedSession = session;
        }
        return cachedSession;
    }

    private Bundle callPsAssistantBridge(String method, Bundle extras) {
        try {
            Context context = AndroidAppHelper.currentApplication();
            if (context == null) return null;
            return context.getContentResolver().call(
                    CarLinkDexContract.BRIDGE_URI,
                    method,
                    null,
                    extras
            );
        } catch (Throwable t) {
            XposedBridge.log(TAG + "Bridge IPC call failed (" + method + "): " + t.getMessage());
            return null;
        }
    }
}
